  "leg swelling":     ["local", "cvs", "msk"],
};

const DEFAULT_SYSTEM_ORDER = [
  "general", "cvs", "resp", "abdomen", "msk", "neuro", "gu", "skin",
];

export function getSystemsForComplaint(complaint) {
  if (!complaint) return ["local", ...DEFAULT_SYSTEM_ORDER];
  const key = complaint.toLowerCase().trim();
  const exact = CC_PRIORITY[key];
  if (exact) return [...exact, ...DEFAULT_SYSTEM_ORDER.filter((s) => !exact.includes(s))];
  // Fuzzy match: any priority key contained in the complaint
  for (const [pkey, systems] of Object.entries(CC_PRIORITY)) {
    if (key.includes(pkey)) {
      return [...systems, ...DEFAULT_SYSTEM_ORDER.filter((s) => !systems.includes(s))];
    }
  }
  return ["local", ...DEFAULT_SYSTEM_ORDER];
}

function Chip({ label, onRemove, style: extraStyle }) {
  return (
    <span style={{ ...chipRemovableStyle, ...(extraStyle || {}) }}>
      {label}
      {onRemove && (
        <button
          onClick={onRemove}
          style={{
            border: "none", background: "transparent",
            cursor: "pointer", padding: 0, marginLeft: 2,
            display: "inline-flex", alignItems: "center",
            color: T.mute,
          }}
          aria-label="Remove"
        >
          <X size={10} />
        </button>
      )}
    </span>
  );
}

function SuggestionChip({ label, onClick }) {
  return (
    <button
      onClick={onClick}
      style={{
        ...chipSuggestionStyle,
        border: "1px dashed " + T.line,
      }}
    >
      <Plus size={9} /> {label}
    </button>
  );
}

export function BackgroundCard({ patient, onUpdate }) {
  const [expanded, setExpanded] = useState(true);
  const [editingSection, setEditingSection] = useState(null);
  const bg = patient || emptyPatientBackground();

  const updateSection = (section, items) => {
    onUpdate({ ...bg, [section]: items });
  };

  const updateSocial = (changes) => {
    onUpdate({ ...bg, social: { ...(bg.social || {}), ...changes } });
  };

  return (
    <div style={cardStyle}>
      <div style={headerRowStyle} onClick={() => setExpanded(!expanded)}>
        <div style={{ display: "flex", alignItems: "center", gap: 8 }}>
          {expanded ? <ChevronDown size={16} color={T.navySoft} /> : <ChevronRight size={16} color={T.navySoft} />}
          <span style={sectionTitleStyle}>Background</span>
          <span style={{ fontSize: 11, color: T.mute, marginLeft: 6 }}>
            (longitudinal, inherited from patient record)
          </span>
        </div>
        <BackgroundSummary bg={bg} />
      </div>

      {expanded && (
        <div style={{ marginTop: 12, display: "flex", flexDirection: "column", gap: 10 }}>
          <BackgroundRow
            label="PMH"
            icon={ClipboardList}
            items={bg.pmh || []}
            suggestions={PMH_CHIPS}
            emptyText="None recorded"
            placeholder="e.g. Hypertension"
            isEditing={editingSection === "pmh"}
            onEdit={() => setEditingSection(editingSection === "pmh" ? null : "pmh")}
            onAdd={(item) => {
              updateSection("pmh", [...(bg.pmh || []), item]);
            }}
            onRemove={(i) => {
              updateSection("pmh", (bg.pmh || []).filter((_, idx) => idx !== i));
            }}
          />

          <BackgroundRow
            label="PSH"
            icon={Stethoscope}
            items={bg.psh || []}
            suggestions={PSH_CHIPS}
            emptyText="No surgical history"
            placeholder="e.g. Cholecystectomy"
            isEditing={editingSection === "psh"}
            onEdit={() => setEditingSection(editingSection === "psh" ? null : "psh")}
            onAdd={(item) => {
              updateSection("psh", [...(bg.psh || []), item]);
            }}
            onRemove={(i) => {
              updateSection("psh", (bg.psh || []).filter((_, idx) => idx !== i));
            }}
          />

          <BackgroundRow
            label="Allergies"
            icon={AlertTriangle}
            items={bg.allergies || []}
            suggestions={ALLERGY_CHIPS}
            emptyText="NKDA"
            emptyColor={T.normalGreen}
            placeholder="e.g. Penicillin"
            isEditing={editingSection === "allergies"}
            onEdit={() => setEditingSection(editingSection === "allergies" ? null : "allergies")}
            onAdd={(item) => {
              updateSection("allergies", [...(bg.allergies || []), item]);
            }}
            onRemove={(i) => {
              updateSection("allergies", (bg.allergies || []).filter((_, idx) => idx !== i));
            }}
            chipStyle={{ background: "#FFF0F0", borderColor: T.crit }}
          />

          <BackgroundRow
            label="Medications"
            icon={Pill}
            items={bg.medications || []}
            suggestions={MEDICATION_CHIPS}
            emptyText="No regular medications"
            placeholder="e.g. Metformin 500mg BD"
            isEditing={editingSection === "medications"}
            onEdit={() => setEditingSection(editingSection === "medications" ? null : "medications")}

              updateSection("medications", [...(bg.medications || []), item]);

function BackgroundSummary({ bg }) {
  const counts = [
    (bg.pmh || []).length && (bg.pmh.length + " PMH"),
    (bg.psh || []).length && (bg.psh.length + " PSH"),
    (bg.allergies || []).length ? (bg.allergies.length + " allergies") : "NKDA",
    (bg.medications || []).length && (bg.medications.length + " meds"),
  ].filter(Boolean);
  if (counts.length === 0) return null;
  return (
    <span style={{ fontSize: 11, color: T.mute }}>
      {counts.join(" \u2022 ")}
    </span>
  );
}

function BackgroundRow({
  label, icon: Icon, items, suggestions, emptyText, emptyColor,
  placeholder, isEditing, onEdit, onAdd, onRemove, chipStyle,
}) {
  const [newItemText, setNewItemText] = useState("");
  const filteredSuggestions = useMemo(() => {
    if (!newItemText) return suggestions.slice(0, 6);
    const t = newItemText.toLowerCase();
    return suggestions.filter((s) => s.toLowerCase().includes(t)).slice(0, 6);
  }, [newItemText, suggestions]);

  const commit = (item) => {
    const v = (item || newItemText).trim();
    if (!v) return;
    onAdd(v);
    setNewItemText("");
  };

  return (
    <div style={{ display: "flex", alignItems: "flex-start", gap: 8 }}>
      <div style={{ ...labelStyle, display: "flex", alignItems: "center", gap: 4 }}>
        {Icon && <Icon size={11} color={T.goldSoft} />}
        <span>{label}</span>
      </div>
      <div style={{ flex: 1 }}>
        <div style={{ display: "flex", flexWrap: "wrap", gap: 4, alignItems: "center" }}>
          {(items || []).length === 0 && !isEditing && (
            <span style={{ fontSize: 11, color: emptyColor || T.mute, fontStyle: "italic" }}>
              {emptyText}
            </span>
          )}
          {(items || []).map((it, i) => (
            <Chip
              key={i}
              label={typeof it === "string" ? it : (it.label || JSON.stringify(it))}
              onRemove={() => onRemove(i)}
              style={chipStyle}
            />
          ))}
          {!isEditing ? (
            <button onClick={onEdit} style={btnSmall}>
              <Plus size={10} /> Add
            </button>
          ) : (
            <button onClick={onEdit} style={btnSmall}>
              <Check size={10} /> Done
            </button>
          )}
        </div>

        {isEditing && (
          <div style={{ marginTop: 6 }}>
            <div style={{ display: "flex", gap: 4, alignItems: "center", marginBottom: 6 }}>
              <input
                type="text"
                value={newItemText}
                onChange={(e) => setNewItemText(e.target.value)}
                placeholder={placeholder}
                style={{ ...inputStyle, flex: 1 }}
                onKeyDown={(e) => { if (e.key === "Enter") commit(); }}
              />
              <button onClick={() => commit()} style={btnPrimarySmall}>
                <Plus size={10} /> Add
              </button>
            </div>
            <div style={{ display: "flex", flexWrap: "wrap", gap: 4 }}>
              {filteredSuggestions.map((s, i) => (
                <SuggestionChip key={i} label={s} onClick={() => commit(s)} />
              ))}
            </div>
          </div>
        )}
      </div>
    </div>
  );
}

function SelectGroup({ label, value, options, onChange }) {
  return (
    <div style={{ display: "flex", flexDirection: "column", gap: 2 }}>
      <span style={{ fontSize: 9, color: T.mute, letterSpacing: 0.5 }}>
        {label.toUpperCase()}
      </span>
      <div style={{ display: "flex", gap: 2 }}>
        {options.map((opt) => (
          <button
            key={opt.value}
            onClick={() => onChange(value === opt.value ? null : opt.value)}
            style={{
              ...btnSmall,
              padding: "2px 6px",
              fontSize: 10,
              background: value === opt.value ? T.navy : T.white,
              color: value === opt.value ? T.white : T.navySoft,

            }}
          >
            {opt.label}
          </button>
        ))}
      </div>
    </div>
  );
}

// 8. SYSTEMS EXAM
// =============================================================

export function SystemsExam({ examination, chiefComplaint, onUpdate }) {
  const exam = examination || {};

  const orderedSystems = useMemo(
    () => getSystemsForComplaint(chiefComplaint),
    [chiefComplaint]
  );

  // Initial expanded state: local + first priority system + any abnormal
  const [expandedSystems, setExpandedSystems] = useState(() => {
    const s = new Set();
    s.add("local");
    if (orderedSystems[1]) s.add(orderedSystems[1]);
    Object.entries(exam).forEach(([sys, data]) => {
      if (data && data.isNormal === false) s.add(sys);
    });
    return s;
  });

  const toggleSystem = (sys) => {
    setExpandedSystems((prev) => {
      const next = new Set(prev);
      if (next.has(sys)) next.delete(sys);
      else next.add(sys);
      return next;
    });
  };

  const updateSystem = (sys, updates) => {
    const current = exam[sys] || { isNormal: true, chips: [], freeText: "" };
    onUpdate({ ...exam, [sys]: { ...current, ...updates } });
  };

  return (
    <div style={cardStyle}>
      <div style={{ ...headerRowStyle, marginBottom: 12, cursor: "default" }}>
        <div style={{ display: "flex", alignItems: "center", gap: 8 }}>
          <Stethoscope size={14} color={T.navySoft} />
          <span style={sectionTitleStyle}>Physical examination</span>
          <span style={{ fontSize: 11, color: T.mute, marginLeft: 6 }}>
            (by system, ordered by chief complaint)
          </span>
        </div>
        <ExamSummary exam={exam} />
      </div>

      <div style={{ display: "flex", flexDirection: "column", gap: 6 }}>
        {orderedSystems.map((sys) => (
          <SystemCard
            key={sys}
            system={sys}
            data={exam[sys] || { isNormal: true, chips: [], freeText: "" }}
            expanded={expandedSystems.has(sys)}
            onToggle={() => toggleSystem(sys)}
            onUpdate={(updates) => updateSystem(sys, updates)}
            availableChips={EXAM_CHIPS[sys] || []}
          />
        ))}
      </div>
    </div>
  );
}

function ExamSummary({ exam }) {
  const abnormalCount = Object.values(exam).filter((d) => d && d.isNormal === false).length;
  const totalDocumented = Object.values(exam).filter(
    (d) => d && (d.isNormal === false || (d.chips || []).length > 0 || (d.freeText || "").trim().length > 0)
  ).length;
  if (totalDocumented === 0) return null;
  return (
    <span style={{ fontSize: 11, color: T.mute }}>
      {totalDocumented} documented
      {abnormalCount > 0 && (
        <span style={{ color: T.abnormalAmber }}>
          {" \u2022 "}{abnormalCount} abnormal
        </span>
      )}
    </span>
  );
}

function SystemCard({ system, data, expanded, onToggle, onUpdate, availableChips }) {
  const info = SYSTEM_INFO[system] || { label: system, icon: Activity, color: T.navySoft };
  const { icon: Icon } = info;

  const selectedChips = data.chips || [];
  const remainingSuggestions = availableChips.filter((c) => !selectedChips.includes(c));

  const documented = !data.isNormal || selectedChips.length > 0 || (data.freeText || "").trim().length > 0;
  const indicatorColor = data.isNormal === false ? T.abnormalAmber : (documented ? T.normalGreen : T.line);

  const handleToggleNormal = () => {
    onUpdate({ isNormal: !data.isNormal });
  };

  const handleAddChip = (chip) => {
    onUpdate({ chips: [...selectedChips, chip] });
  };

  const handleRemoveChip = (chip) => {
    onUpdate({ chips: selectedChips.filter((c) => c !== chip) });
  };

  const handleFreeTextChange = (e) => {
    onUpdate({ freeText: e.target.value });
  };

  return (
    <div style={{
      border: "1px solid " + T.line,
      borderLeft: "3px solid " + indicatorColor,
      borderRadius: 4,
      background: T.card,
    }}>
      <div
        onClick={onToggle}
        style={{
          ...headerRowStyle,
          padding: "8px 10px",
        }}
      >
        <div style={{ display: "flex", alignItems: "center", gap: 8 }}>
          {expanded ? <ChevronDown size={14} color={T.navySoft} /> : <ChevronRight size={14} color={T.navySoft} />}
          <Icon size={13} color={info.color} />
          <span style={{ fontSize: 13, fontWeight: 600, color: T.navy }}>
            {info.label}
          </span>
          {data.isNormal === false && (
            <span style={{
              fontSize: 9, fontWeight: 700,
              color: T.abnormalAmber,
              textTransform: "uppercase",
              letterSpacing: 0.8,
              background: T.warnBg,
              padding: "1px 5px",
              borderRadius: 2,
            }}>
              Abnormal
            </span>
          )}
        </div>
        <span style={{ fontSize: 10, color: T.mute }}>
          {selectedChips.length > 0 && (selectedChips.length + " findings")}
          {selectedChips.length === 0 && data.isNormal && "Normal"}
        </span>
      </div>

      {expanded && (
        <div style={{
          padding: "0 10px 10px 10px",
          display: "flex", flexDirection: "column", gap: 8,
        }}>
          <div style={{ display: "flex", alignItems: "center", gap: 6 }}>
            <button
              onClick={handleToggleNormal}
              style={{
                ...btnSmall,
                background: data.isNormal ? T.normalGreen : T.white,
                color: data.isNormal ? T.white : T.navySoft,
                border: data.isNormal ? "none" : "1px solid " + T.line,
              }}
            >
              <Check size={10} /> Normal
            </button>
            <button
              onClick={handleToggleNormal}
              style={{
                ...btnSmall,
                background: data.isNormal === false ? T.abnormalAmber : T.white,
                color: data.isNormal === false ? T.white : T.navySoft,
                border: data.isNormal === false ? "none" : "1px solid " + T.line,
              }}
            >
              <AlertTriangle size={10} /> Abnormal
            </button>
          </div>

          {/* Selected findings */}
          {selectedChips.length > 0 && (
            <div style={{ display: "flex", flexWrap: "wrap", gap: 4 }}>
              {selectedChips.map((chip, i) => (
                <Chip key={i} label={chip} onRemove={() => handleRemoveChip(chip)} />
              ))}
            </div>
          )}

          {/* Available chips */}
          {remainingSuggestions.length > 0 && (
            <div>
              <div style={{
                fontSize: 9, color: T.mute,
                textTransform: "uppercase", letterSpacing: 0.5,
                marginBottom: 4,
              }}>
                Common findings
              </div>
              <div style={{ display: "flex", flexWrap: "wrap", gap: 4 }}>
                {remainingSuggestions.map((chip, i) => (
                  <SuggestionChip key={i} label={chip} onClick={() => handleAddChip(chip)} />
                ))}
              </div>
            </div>
          )}

          {/* Free text overflow */}
          <div>
            <div style={{
              fontSize: 9, color: T.mute,
              textTransform: "uppercase", letterSpacing: 0.5,
              marginBottom: 4,
            }}>
              Additional notes
            </div>
            <textarea
              value={data.freeText || ""}
              onChange={handleFreeTextChange}
              placeholder="Free text for findings not covered by chips above..."
              rows={2}
              style={{
                ...inputStyle,
                width: "100%",
                resize: "vertical",
                fontFamily: "inherit",
                boxSizing: "border-box",
              }}
            />
          </div>
        </div>
      )}
    </div>
  );
}
